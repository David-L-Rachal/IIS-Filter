void PopulateUserMap()
{
    CADOBinding conn;
    int iLoop = 0;
    int iCount = 0;
    if (conn.Open("SQLOLEDB", szServer, szDb, szUid, szPwd))
    {        
        CDataReturn data;

        if(conn.Execute(data.SQL().c_str(), &data, &data))
        {
            
            {
                iCount = conn.RecordCount();
                if(iCount > 1)
                {
                    
                    EnterCriticalSection(&cs);    
                    RtlZeroMemory(szBuffer, 1000);
                    lstrcpy(szBuffer, data.UserAndPath.Value);
                    if (lstrcmp(szLastUpdate, szBuffer) != 0)
                    {
                                                                        
                        RtlZeroMemory(szLastUpdate, 1000);
                        lstrcpy(szLastUpdate, data.UserAndPath.Value);
                        
                        conn.MoveNext();

                        if( aSwitch == 0)
                        {
                            smCache1.FreeArray();
                            smCache1.MakeArray(iCount - 1);                         
                            do
                            {
                                
                                RtlZeroMemory(szBuffer, 1000);
                                lstrcpy(szBuffer, data.UserAndPath.Value);
                                OutputDebugString(szBuffer);
                                smCache1.SetItem(iLoop, data.UserAndPath.Value);
                                iLoop++;
                            }while(conn.MoveNext());
                            aSwitch = 1;
                               
                        }
                        else
                        {
                            smCache0.FreeArray();
                            smCache0.MakeArray(iCount - 1);                         
                            do
                            {

                                RtlZeroMemory(szBuffer, 1000);
                                lstrcpy(szBuffer, data.UserAndPath.Value);
                                OutputDebugString(szBuffer);
                                smCache0.SetItem(iLoop, data.UserAndPath.Value);
                                iLoop++;
                            }while(conn.MoveNext());

                            aSwitch = 0;
                        }         
                                         
                    }  
                    
                    smFreeResources.MakeArray(4);

                    smFreeResources.SetItem(0, "/faq.html");
                    smFreeResources.SetItem(1, "/favicon.ico");
                    smFreeResources.SetItem(2, "/music2.mp3");
                    smFreeResources.SetItem(3, "/subfolder1/test.asp");
                   
                    LeaveCriticalSection(&cs);    
                }
            
            }       
        
        }
        
        conn.Close();
    }

}

bool IsValidUser(char* codedPattern, char* szLocation)
{

    allocSize = lstrlen(codedPattern) + lstrlen(szLocation) + 1;

    char szResource[256];

    lstrcpy(szCriteria, codedPattern);
    strlwr(szLocation);
    lstrcat(szCriteria, szLocation);

    OutputDebugString(szCriteria);

    EnterCriticalSection(&cs);

    if (smFreeResources.Find(szLocation).start >= 0)
    {
        lstrcpy(szResource, "-- free resource: ");
        lstrcat(szResource, szLocation);

    }
    else
    {

        if (foundUser == false)

DWORD WINAPI __stdcall HttpFilterProc(HTTP_FILTER_CONTEXT *pfc, DWORD dwNotificationType, LPVOID *lpNotificationData)
{
    if (dwNotificationType == SF_NOTIFY_PREPROC_HEADERS)
    {
        HTTP_FILTER_PREPROC_HEADERS* fpH;
        
        char szRetHead[MAX_LEN + 1];
        char szUrlLoc[(MAX_LEN * 2) + 1] = {0};
        
		//char szNewCookie[5] = {0};
		//char szBuff1[MAX_LEN] = {0};
		//char szBuff2[MAX_LEN] = {0};
        //char szClortho[82] = {0};
        unsigned long ulLenHead = MAX_LEN;
        unsigned long ulLenUrl = MAX_LEN * 2;
        fpH = (HTTP_FILTER_PREPROC_HEADERS*)lpNotificationData;
        
		//szNewCookie = (char*)malloc(5);
		//szBuff1 = (char*)malloc(MAX_LEN);
		//szBuff2 = (char*)malloc(MAX_LEN);
		//RtlZeroMemory(szClortho, 82);
		//RtlZeroMemory(szNewCookie, 5);
        RtlZeroMemory(szUrlLoc, (MAX_LEN * 2) + 1);
        RtlZeroMemory(szRetHead, MAX_LEN + 1);
       
        if (fpH->GetHeader(pfc, (char*)FLTR_AUTH, szRetHead, &ulLenHead) == TRUE && fpH->GetHeader(pfc, (char*)FLTR_PATH, szUrlLoc, &ulLenUrl) == TRUE)
        {
            char* szQueryStringLocation = NULL;
            szQueryStringLocation = strstr(szUrlLoc, "?");
            if (szQueryStringLocation != NULL)
            {
                szQueryStringLocation[0] = NULL;
            }

            if(IsValidUser(szRetHead, szUrlLoc))
            {   
                OutputDebugString("!! passed");
            }
            else
            {
                OutputDebugString("** failed authentication");
                pfc->ServerSupportFunction(pfc, SF_REQ_ADD_HEADERS_ON_DENIAL, (void*)FLTR_HEAD, NULL, NULL);
                SetLastError(ERROR_ACCESS_DENIED);
             
				return SF_STATUS_REQ_ERROR;
            }       
        }
        else //No header of error so reject them
        {
            if (GetLastError() == ERROR_INSUFFICIENT_BUFFER)
            {
                OutputDebugString("** buffer too small");
                SetLastError(ERROR_ACCESS_DENIED);
            }
            else
            {
                OutputDebugString("** failed with no header");
                pfc->ServerSupportFunction(pfc, SF_REQ_ADD_HEADERS_ON_DENIAL, (void*)FLTR_HEAD, NULL, NULL);
                SetLastError(ERROR_ACCESS_DENIED);
            }
            
			return SF_STATUS_REQ_ERROR;
        }
    }


    return SF_STATUS_REQ_NEXT_NOTIFICATION;
}
